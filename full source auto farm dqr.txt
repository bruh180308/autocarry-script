local spoofUpdateGuard = setmetatable({}, { __mode = "k" })
local spoofObserved = setmetatable({}, { __mode = "k" })
local function replacePlain(text, needle, replacement)
    local pattern = tostring(needle):gsub("(%W)", "%%%1")
    return tostring(text):gsub(pattern, function() return replacement end)
end

local function applyNameSpoof(inst)
    if not runtimeActive() then return end
    if not (inst:IsA("TextLabel") or inst:IsA("TextButton")) then return end

    local function updateText()
        if not runtimeActive() then return end
        if spoofUpdateGuard[inst] then return end
        local cur = inst.Text
        if not cur or cur == "" then return end
        local newText = cur
        for rName, _ in pairs(allRealNamesSet) do
            if #rName >= 2 and string.find(newText, rName, 1, true) then
                newText = replacePlain(newText, rName, SPOOFED_NAME)
            end
        end
        if newText ~= cur then
            spoofUpdateGuard[inst] = true
            inst.Text = newText
            spoofUpdateGuard[inst] = nil
        end
    end

    updateText()
    if spoofObserved[inst] then return end
    spoofObserved[inst] = true
    inst:GetPropertyChangedSignal("Text"):Connect(updateText)
end

local function updateSpoofedName(newName)
    newName = tostring(newName or ""):match("^%s*(.-)%s*$")
    if newName == "" or newName == SPOOFED_NAME then return false end
    local previous = SPOOFED_NAME
    SPOOFED_NAME = newName
    local roots = { localPlayer:FindFirstChildOfClass("PlayerGui"), workspace }
    pcall(function() table.insert(roots, game:GetService("CoreGui")) end)
    for _, root in ipairs(roots) do
        if root then
            for _, inst in ipairs(root:GetDescendants()) do
                if (inst:IsA("TextLabel") or inst:IsA("TextButton")) and string.find(inst.Text, previous, 1, true) then
                    spoofUpdateGuard[inst] = true
                    inst.Text = replacePlain(inst.Text, previous, newName)
                    spoofUpdateGuard[inst] = nil
                end
            end
        end
    end
    return true
end

local function getDungeonProgress()
    local progress = workspace:FindFirstChild("dungeonProgress")
    if not progress then return nil end
    local value = tostring(progress.Value)
    local normalized = string.lower((value:match("^%s*(.-)%s*$")))
    local compact = normalized:gsub("[%s_%-]", "")
    if compact == "inprocess" or compact == "inprogress" then
        return "in process"
    end
    if compact == "bosskilled" then
        return "bosskilled"
    end
    return normalized
end

local matchStartedTimestamp = isMatchStarted() and os.clock() or nil
local matchCycle = {
    forceNewMatch = false,
    sawActiveProgress = false,
    finishConsumed = false,
}

local function armNewMatchCycle()
    matchCycle.forceNewMatch = true
    matchCycle.sawActiveProgress = false
    matchCycle.finishConsumed = false
end

local function observeMatchState()
    local started = isInDungeon()
    local entered = started and (matchStartedTimestamp == nil or matchCycle.forceNewMatch)
    if entered then
        matchStartedTimestamp = os.clock()
        matchCycle.forceNewMatch = false
        matchCycle.sawActiveProgress = false
        matchCycle.finishConsumed = false
    elseif not started then
        matchStartedTimestamp = nil
        matchCycle.sawActiveProgress = false
        matchCycle.finishConsumed = false
    end

    local progress = getDungeonProgress()
    if started and progress == "in process" then
        matchCycle.sawActiveProgress = true
        matchCycle.finishConsumed = false
    end
    return started, entered
end

local function isDungeonFinished()
    local started = observeMatchState()
    if not started then return false end

    if matchCycle.finishConsumed then return false end
    if getDungeonProgress() ~= "bosskilled" then return false end

    matchCycle.finishConsumed = true
    return true
end

local reportedRemoteProblems = {}
local function reportRemoteProblem(name, message)
    local key = name .. ":" .. message
    if reportedRemoteProblems[key] then return end
    reportedRemoteProblems[key] = true
    warn(string.format("[AutoCarry] Remote %s: %s", name, message))
end

local function getRemote(name, expectedClass)
    if not runtimeActive() then return nil end
    if not remotes or not remotes.Parent then
        remotes = ReplicatedStorage:FindFirstChild("remotes") or ReplicatedStorage:WaitForChild("remotes", 5)
    end
    local remote = remotes and (remotes:FindFirstChild(name) or remotes:WaitForChild(name, 2))
    if not remote then
        reportRemoteProblem(name, "not found")
        return nil
    end
    if expectedClass and not remote:IsA(expectedClass) then
        reportRemoteProblem(name, "unexpected class (" .. remote.ClassName .. ")")
        return nil
    end
    return remote
end

local function fireRemote(name, ...)
    local remote = getRemote(name, "RemoteEvent")
    if not remote then return false end
    local args = table.pack(...)
    local ok, err = pcall(function()
        remote:FireServer(table.unpack(args, 1, args.n))
    end)
    if not ok then reportRemoteProblem(name, tostring(err)) end
    return ok
end

local function invokeRemote(name, ...)
    local remote = getRemote(name, "RemoteFunction")
    if not remote then return false, nil end
    local args = table.pack(...)
    local ok, result = pcall(function()
        return remote:InvokeServer(table.unpack(args, 1, args.n))
    end)
    if not ok then reportRemoteProblem(name, tostring(result)) end
    return ok, result
end

local function warriorItemScore(item)
    return tonumber(item.physicalDamage) or 0
end

local function mageItemScore(item)
    return math.max(tonumber(item.physicalPower) or 0, tonumber(item.spellPower) or 0)
end

local function rarityScore(rarity)
    local scores = { common = 1, uncommon = 2, rare = 3, epic = 4, legendary = 5, ultimate = 6 }
    return scores[string.lower(tostring(rarity or ""))] or 0
end

local function isBetterBuildItem(candidate, candidateKey, current, currentKey, scoreFn)
    if not current then return true end
    local candidateScore, currentScore = scoreFn(candidate), scoreFn(current)
    if candidateScore ~= currentScore then return candidateScore > currentScore end
    local candidateLevel, currentLevel = getItemLevelReq(candidate), getItemLevelReq(current)
    if candidateLevel ~= currentLevel then return candidateLevel > currentLevel end
    local candidateUpgrade = tonumber(candidate.currentUpgrade) or 0
    local currentUpgrade = tonumber(current.currentUpgrade) or 0
    if candidateUpgrade ~= currentUpgrade then return candidateUpgrade > currentUpgrade end
    local candidateRarity, currentRarity = rarityScore(candidate.rarity), rarityScore(current.rarity)
    if candidateRarity ~= currentRarity then return candidateRarity > currentRarity end
    return tostring(candidateKey) < tostring(currentKey)
end

local function getBestGearForBuild(bucket, scoreFn)
    local bestItem, bestKey = nil, nil
    local playerLevel = getLocalLevel()
    if playerLevel <= 0 then playerLevel = math.huge end
    for key, item in pairs(bucket or {}) do
        if typeof(item) == "table" and getItemLevelReq(item) <= playerLevel and isBetterBuildItem(item, key, bestItem, bestKey, scoreFn) then
            bestItem, bestKey = item, key
        end
    end
    return bestItem, bestKey
end

local function abilityMatchesBuild(item, build)
    local description = string.lower(tostring(item.description or ""))
    if build == "warrior" then
        return string.find(description, "scales with physical damage", 1, true) ~= nil
    end
    return string.find(description, "scales with spell power", 1, true) ~= nil
        or string.find(description, "scales with physical power", 1, true) ~= nil
end

local function isUniversalHeal(item)
    return string.find(string.lower(tostring(item and item.name or "")), "universal heal", 1, true) ~= nil
end

local function getAutoLoadout(inventory)
    local warriorGear, mageGear = {}, {}
    local warriorTotal, mageTotal = 0, 0
    local gearTypes = {
        { singular = "weapon", plural = "weapons" },
        { singular = "chest", plural = "chests" },
        { singular = "helmet", plural = "helmets" },
    }

    for _, gearType in ipairs(gearTypes) do
        local bucket = inventory[gearType.plural] or {}
        local warriorItem, warriorKey = getBestGearForBuild(bucket, warriorItemScore)
        local mageItem, mageKey = getBestGearForBuild(bucket, mageItemScore)
        warriorGear[gearType.singular] = warriorItem and { item = warriorItem, key = warriorKey } or nil
        mageGear[gearType.singular] = mageItem and { item = mageItem, key = mageKey } or nil
        warriorTotal = warriorTotal + (warriorItem and warriorItemScore(warriorItem) or 0)
        mageTotal = mageTotal + (mageItem and mageItemScore(mageItem) or 0)
    end

    local build = string.lower(tostring(CONFIG.AUTO_EQUIP_BUILD or "mage"))
    if build ~= "warrior" and build ~= "mage" then build = "mage" end

    local healCandidates, abilityCandidates = {}, {}
    local playerLevel = getLocalLevel()
    if playerLevel <= 0 then playerLevel = math.huge end
    for key, item in pairs(inventory.abilities or {}) do
        if typeof(item) == "table" and getItemLevelReq(item) <= playerLevel then
            local entry = { item = item, key = key }
            if not isHost and isUniversalHeal(item) then
                table.insert(healCandidates, entry)
            elseif abilityMatchesBuild(item, build) then
                table.insert(abilityCandidates, entry)
            end
        end
    end

    local function sortAbilities(list)
        table.sort(list, function(a, b)
            local aLevel, bLevel = getItemLevelReq(a.item), getItemLevelReq(b.item)
            if aLevel ~= bLevel then return aLevel > bLevel end
            local aRarity, bRarity = rarityScore(a.item.rarity), rarityScore(b.item.rarity)
            if aRarity ~= bRarity then return aRarity > bRarity end
            return tostring(a.key) < tostring(b.key)
        end)
    end
    sortAbilities(healCandidates)
    sortAbilities(abilityCandidates)

    local chosenAbilities, selectedKeys = {}, {}
    local function addAbility(entry)
        if not entry or selectedKeys[entry.key] or #chosenAbilities >= 4 then return end
        selectedKeys[entry.key] = true
        table.insert(chosenAbilities, entry)
    end
    for index = 1, math.min(2, #healCandidates) do addAbility(healCandidates[index]) end
    for _, entry in ipairs(abilityCandidates) do addAbility(entry) end

    return {
        build = build,
        gear = build == "warrior" and warriorGear or mageGear,
        warriorTotal = warriorTotal,
        mageTotal = mageTotal,
        abilities = chosenAbilities,
    }
end

local function makeLoadoutReservation(loadout)
    local reserved = { weapons = {}, chests = {}, helmets = {}, abilities = {} }
    for singular, entry in pairs(loadout.gear) do
        if entry then
            reserved[singular .. "s"][entry.key] = true
        end
    end
    for _, entry in ipairs(loadout.abilities) do
        reserved.abilities[entry.key] = true
    end
    return reserved
end

local autoEquipRunning = false
local inventoryAutomationBusy = false
local lastAutoEquipAt = -math.huge
local inventoryRunToken = nil
local function beginInventoryRun(kind)
    if inventoryAutomationBusy then return nil end
    local token = {}
    inventoryRunToken = token
    inventoryAutomationBusy = true
    autoEquipRunning = kind == "equip"
    autoSellRunning = kind == "sell"
    task.delay(20, function()
        if inventoryRunToken == token then
            inventoryRunToken = nil
            inventoryAutomationBusy = false
            autoEquipRunning = false
            autoSellRunning = false
            if kind == "equip" then
                nextAutoEquipAllowedAt = os.clock() + 30
            else
                nextAutoSellAllowedAt = os.clock() + 30
            end
            if runtimeActive() and UI then UI.addLog("Inventory operation timed out and was released", "warn") end
        end
    end)
    return token
end

local function finishInventoryRun(token)
    if inventoryRunToken ~= token then return end
    inventoryRunToken = nil
    inventoryAutomationBusy = false
    autoEquipRunning = false
    autoSellRunning = false
end

local function runAutoEquip()
    if isHost or not runtimeActive() or autoEquipRunning or inventoryAutomationBusy or os.clock() < nextAutoEquipAllowedAt then return end
    if isInDungeon() then
        local char = localPlayer.Character
        local busy = char and char:FindFirstChild("busyCasting")
        if busy and busy.Value == true then return end
    end
    local runToken = beginInventoryRun("equip")
    if not runToken then return end
    if UI then UI.setFlow("OPTIMIZING LOADOUT", Color3.fromRGB(168, 85, 247)) end

    local ok, inventory = invokeRemote("reloadInvy")
    if not runtimeActive() or not ok or typeof(inventory) ~= "table" then
        nextAutoEquipAllowedAt = os.clock() + 30
        finishInventoryRun(runToken)
        return
    end

    local loadout = getAutoLoadout(inventory)
    local changes = 0
    local failed = false
    local function changeLoadout(remoteName, ...)
        local called, accepted = invokeRemote(remoteName, ...)
        local success = called and accepted ~= false
        if not success then failed = true end
        return success
    end
    local gearInventoryKeys = { weapon = "weapons", chest = "chests", helmet = "helmets" }

    for itemType, plural in pairs(gearInventoryKeys) do
        if not runtimeActive() then break end
        local target = loadout.gear[itemType]
        if target then
            local equippedKey = nil
            for key, item in pairs(inventory[plural] or {}) do
                if typeof(item) == "table" and item.equipped ~= false then
                    equippedKey = key
                    break
                end
            end
            if equippedKey ~= target.key then
                local canEquipTarget = true
                if equippedKey then
                    local oldId = getInventoryItemNumericId(equippedKey)
                    if not oldId or not changeLoadout("unequipItem", itemType, oldId) then
                        canEquipTarget = false
                    else
                        task.wait(CONFIG.EQUIP_REMOTE_DELAY)
                    end
                end
                local targetId = getInventoryItemNumericId(target.key)
                if canEquipTarget and targetId then
                    local equipped = changeLoadout("equipItem", itemType, targetId)
                    if equipped then
                        changes = changes + 1
                        task.wait(CONFIG.EQUIP_REMOTE_DELAY)
                    end
                end
            end
        end
    end

    if #loadout.abilities > 0 then
        local desiredSlots = { "q", "e", "q2", "e2" }
        local currentSlots = {}
        for key, item in pairs(inventory.abilities or {}) do
            if typeof(item) == "table" and typeof(item.equipped) == "table" then
                for _, slot in ipairs(desiredSlots) do
                    if item.equipped[slot] == true then currentSlots[slot] = key end
                end
            end
        end

        local desiredBySlot = {}
        for index, entry in ipairs(loadout.abilities) do
            desiredBySlot[desiredSlots[index]] = entry.key
        end

        local clearedAbilityKeys = {}
        for _, slot in ipairs(desiredSlots) do
            if not runtimeActive() then break end
            local currentKey = currentSlots[slot]
            local desiredKey = desiredBySlot[slot]
            if desiredKey and currentKey and currentKey ~= desiredKey then
                if clearedAbilityKeys[currentKey] then
                    currentSlots[slot] = nil
                else
                    local oldId = getInventoryItemNumericId(currentKey)
                    if oldId and changeLoadout("unequipItem", "ability", oldId) then
                        clearedAbilityKeys[currentKey] = true
                        for _, occupiedSlot in ipairs(desiredSlots) do
                            if currentSlots[occupiedSlot] == currentKey then currentSlots[occupiedSlot] = nil end
                        end
                        task.wait(CONFIG.EQUIP_REMOTE_DELAY)
                    end
                end
            end
        end

        for _, slot in ipairs(desiredSlots) do
            if not runtimeActive() then break end
            local targetKey = desiredBySlot[slot]
            if targetKey and currentSlots[slot] ~= targetKey then
                local targetId = getInventoryItemNumericId(targetKey)
                if targetId then
                    local equipped = changeLoadout("equipItem", "ability", targetId, slot)
                    if equipped then
                        currentSlots[slot] = targetKey
                        changes = changes + 1
                        task.wait(CONFIG.EQUIP_REMOTE_DELAY)
                    end
                end
            end
        end
    end

    if changes > 0 and UI then
        UI.addLog("Loadout optimized: " .. string.upper(loadout.build) .. " (" .. tostring(changes) .. " changes)", "success")
    end
    nextAutoEquipAllowedAt = os.clock() + (failed and 30 or CONFIG.AUTO_EQUIP_INTERVAL)
    finishInventoryRun(runToken)
end

local function runAutoLevelSell()
    if isHost or not runtimeActive() or autoSellRunning or inventoryAutomationBusy or os.clock() < nextAutoSellAllowedAt then return end
    local runToken = beginInventoryRun("sell")
    if not runToken then return end
    if UI then UI.setFlow("CLEANING INVENTORY", Color3.fromRGB(244, 63, 94)) end

    local ok, inventory = invokeRemote("reloadInvy")
    if not runtimeActive() or not ok or typeof(inventory) ~= "table" then
        nextAutoSellAllowedAt = os.clock() + 30
        finishInventoryRun(runToken)
        return
    end

    local reserved = makeLoadoutReservation(getAutoLoadout(inventory))
    local payload = { helmet = {}, ability = {}, chest = {}, weapon = {} }
    local sellCount = 0

    for _, info in ipairs(AUTO_SELL_TYPES) do
        local bucket = inventory[info.inventoryKey] or {}
        local highestReq = -1
        for _, item in pairs(bucket) do
            if typeof(item) == "table" then
                highestReq = math.max(highestReq, getItemLevelReq(item))
            end
        end

        for key, item in pairs(bucket) do
            if typeof(item) == "table" then
                local keep = isInventoryItemEquipped(info.payloadKey, item)
                    or getItemLevelReq(item) == highestReq
                    or reserved[info.inventoryKey][key] == true
                    or (not isHost and info.payloadKey == "ability" and isUniversalHeal(item))
                    or item.locked == true
                    or item.favorite == true
                    or rarityScore(item.rarity) >= rarityScore("legendary")
                if not keep then
                    local itemId = getInventoryItemNumericId(key)
                    if itemId then
                        table.insert(payload[info.payloadKey], itemId)
                        sellCount = sellCount + 1
                    end
                end
            end
        end
    end

    if sellCount > 0 then
        local sent = fireRemote("sellItemEvent", payload)
        if sent and UI then
            UI.addLog("Inventory cleaned: " .. tostring(sellCount) .. " low-level items sold", "info")
            task.wait(0.8)
        end
    end
    nextAutoSellAllowedAt = os.clock() + CONFIG.AUTO_SELL_INTERVAL
    finishInventoryRun(runToken)
end

local function startInventoryTask(kind, callback)
    task.spawn(function()
        if not runtimeActive() then return end
        local ok, err = xpcall(callback, function(problem) return tostring(problem) end)
        if ok then return end
        local token = inventoryRunToken
        if token then finishInventoryRun(token) end
        if kind == "equip" then
            nextAutoEquipAllowedAt = os.clock() + 30
        else
            nextAutoSellAllowedAt = os.clock() + 30
        end
        reportRemoteProblem("inventory:" .. kind, err)
        if runtimeActive() and UI then UI.addLog("Inventory " .. kind .. " failed and was released", "warn") end
    end)
end

local function createStageLobby(mapName, difficulty)
    if not isInLobby() then return false, "not_in_lobby" end

    local args = { mapName, difficulty, 0, true, true, false }
    local ok, result = invokeRemote("createLobby", table.unpack(args))
    if not ok or result ~= true then return false, result end

    pcall(function()
        local playerGui = localPlayer:FindFirstChildOfClass("PlayerGui")
        local queueGui = playerGui and playerGui:FindFirstChild("queueGui")
        local lobbyInfo = queueGui and queueGui:FindFirstChild("lobbyInfo")
        local chooser = queueGui and queueGui:FindFirstChild("chooseDungeon")
        if lobbyInfo then
            local startBackground = lobbyInfo:FindFirstChild("startBackground")
            local whitelist = lobbyInfo:FindFirstChild("whitelist")
            if startBackground then startBackground.Visible = true end
            if whitelist then whitelist.Visible = true end
            lobbyInfo.Visible = true
        end
        if chooser then chooser.Visible = false end
    end)
    return true, result
end

local lastEnterDungeonAttempt = -math.huge
local function enterDungeon()
    if not isInLobby() or os.clock() - lastEnterDungeonAttempt < CONFIG.START_COOLDOWN then return false end
    lastEnterDungeonAttempt = os.clock()
    if UI then UI.addLog("Entering dungeon", "info") end
    return fireRemote("startDungeon")
end

local lastStartAttempt = -math.huge
local canHostStartMatch = function() return true end
local function startMatch()
    if isInLobby() then return enterDungeon() end
    local progress = getDungeonProgress()
    if progress == "in process" or progress == "bosskilled" then return false end
    if isHost then
        local allowed, reason = canHostStartMatch()
        if not allowed then
            if UI then UI.addLog("Start delayed: " .. tostring(reason or "clone not ready"), "warn") end
            return false
        end
    end
    if os.clock() - lastStartAttempt < CONFIG.START_COOLDOWN then return false end
    lastStartAttempt = os.clock()
    if UI then UI.addLog("Starting stage", "info") end
    local started = fireRemote("changeStartValue")
    if started then armNewMatchCycle() end
    return started
end

local isLeaving = false
local function leaveGame(reason)
    if not runtimeActive() or isLeaving then return end
    isLeaving = true
    if not isHost and isInDungeon() then
        local refreshed = false
        task.spawn(function()
            pcall(refreshGamepassPricesForHistory)
            refreshed = true
        end)
        local refreshDeadline = os.clock() + 5
        repeat task.wait(0.05) until refreshed or os.clock() >= refreshDeadline or not runtimeActive()
        if not runtimeActive() then isLeaving = false; return false end
        recordGamepassPriceHistory(reason)
    end
    if UI then UI.addLog("Returning to lobby: " .. (reason or "Automatic"), "warn") end
    local remoteSent = fireRemote("ReturnToLobbyEvent")

    local waitDeadline = os.clock() + 5
    repeat
        task.wait(0.5)
    until isInLobby() or os.clock() >= waitDeadline or not runtimeActive()

    if not runtimeActive() then
        isLeaving = false
        return false
    end

    if isInLobby() then
        isLeaving = false
        return true
    end

    local ok, err = pcall(function() TeleportService:Teleport(lobbyPlaceId, localPlayer) end)
    if not ok then
        isLeaving = false
        return remoteSent
    else
        task.delay(10, function() isLeaving = false end)
        return true
    end
end

local function findAbilityTool(slot)
    local containers = { localPlayer:FindFirstChildOfClass("Backpack"), localPlayer.Character }
    for _, container in ipairs(containers) do
        if container then
            for _, item in ipairs(container:GetChildren()) do
                local abilitySlot = item:IsA("Tool") and item:FindFirstChild("abilitySlot")
                if abilitySlot and tostring(abilitySlot.Value) == slot then return item end
            end
        end
    end
    return nil
end

local nextAbilityCastAt = { q = -math.huge, e = -math.huge }
local function castAbilitySafely(slot)
    if isHost then return false end
    local now = os.clock()
    if now < nextAbilityCastAt[slot] or manualCombatIsActive() then return false end
    local char = localPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    local busy = char and char:FindFirstChild("busyCasting")
    if not humanoid or humanoid.Health <= 0 or (busy and busy.Value == true) then return false end
    local tool = findAbilityTool(slot)
    if not tool then return false end
    local cooldown = tool:FindFirstChild("cooldown")
    if cooldown and tonumber(cooldown.Value) and tonumber(cooldown.Value) > 0 then return false end
    local actionRemote = tool:FindFirstChild("abilityEvent") or tool:FindFirstChild("spellEvent")
    local abilityUsed = getRemote("abilityUsed", "RemoteEvent")
    if not actionRemote or not actionRemote:IsA("RemoteEvent") or not abilityUsed then return false end
    nextAbilityCastAt[slot] = now + CONFIG.AUTO_SKILL_RETRY_DELAY
    local ok, err = pcall(function()
        actionRemote:FireServer()
        abilityUsed:FireServer(slot, tool)
    end)
    if not ok then reportRemoteProblem("autoSkill:" .. slot, tostring(err)) end
    return ok
end

task.spawn(function()

        if isHost or not autoToggle or not autoSkillEnabled or isInLobby() or inventoryAutomationBusy or manualCombatIsActive() then continue end
        castAbilitySafely(slots[slotIndex])
        slotIndex = slotIndex == #slots and 1 or slotIndex + 1
    end
end)

local function getPlayerLevel(plr)
    if not plr then return 0 end
    for _, attr in ipairs({ "Level", "level", "Lvl" }) do
        local val = tonumber(plr:GetAttribute(attr))
        if val then return val end
    end
    local ls = plr:FindFirstChild("leaderstats")
    local lvlObj = ls and ls:FindFirstChild("Level")
    if lvlObj and tonumber(lvlObj.Value) then return tonumber(lvlObj.Value) end
    return 0
end

local cloneMemoryState = {
    lastFile = "AutoCarry_LastDungeonClones.json",
    remembered = {},
}

function cloneMemoryState.load()
    cloneMemoryState.remembered = {}
    local ok, decoded = pcall(function()
        if isfile and isfile(cloneMemoryState.lastFile) and readfile then
            return HttpService:JSONDecode(readfile(cloneMemoryState.lastFile))
        end
    end)
    if ok and type(decoded) == "table" and type(decoded.clones) == "table" then
        for _, record in ipairs(decoded.clones) do
            if type(record) == "table" then
                local name = tostring(record.name or ""):match("^%s*(.-)%s*$")
                if name ~= "" then
                    cloneMemoryState.remembered[string.lower(name)] = { name = name, level = math.max(0, tonumber(record.level) or 0) }
                end
            end
        end
    end
end

function cloneMemoryState.save()
    pcall(function()
        if not writefile then return end
        local list = {}
        for _, v in pairs(cloneMemoryState.remembered) do table.insert(list, v) end
        table.sort(list, function(a, b) return tostring(a.name) < tostring(b.name) end)
        writefile(cloneMemoryState.lastFile, HttpService:JSONEncode({ clones = list, stage = currentStage }))
    end)
end

function cloneMemoryState.lowest()
    local list = {}
    for _, v in pairs(cloneMemoryState.remembered) do table.insert(list, v) end
    table.sort(list, function(a, b) return (tonumber(a.level) or 0) < (tonumber(b.level) or 0) end)
    return list[1], #list
end

local function getPlayerByNameInsensitive(name)
    local wanted = string.lower(tostring(name or ""))
    if wanted == "" then return nil end
    for _, plr in ipairs(Players:GetPlayers()) do
        local rName = (playerRealNames and playerRealNames[plr]) or tostring(plr.Name)
        if string.lower(rName) == wanted then return plr end
    end
    return nil
end

local function getReadyDungeonCloneCounts()
    local ready, connected = 0, 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer then
            connected = connected + 1
            local char = plr.Character
            if char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChildOfClass("Humanoid")) then
                ready = ready + 1
            end
        end
    end
    return ready, connected
end

local function getStageForCloneLevel(level)
    local selected = 1
    for index, stage in ipairs(STAGES) do
        if level >= stage.minLvl then selected = index else break end
    end
    return selected
end

local function syncStageFromLowestCloneLevel(lowestLevel)
    if not isHost or not lowestLevel or lowestLevel <= 0 then return end
    local desiredStage = getStageForCloneLevel(lowestLevel)
    if currentStage ~= desiredStage then
        currentStage = desiredStage
        UI.setStage(currentStage, lowestLevel)
        saveStage()
        UI.addLog(string.format("Lowest clone LV.%d selected stage %d: %s (%s)", lowestLevel, desiredStage, STAGES[desiredStage].map, STAGES[desiredStage].diff), "success")
    end
end

function cloneMemoryState.runHost()
    cloneMemoryState.load()

    local hostState = {
        isReplaying = false,
        lastJoinActivityAt = os.clock(),
        lastLobbyCreateAt = -math.huge,
        lobbyCreatedSuccessfully = false,
        lobbyStartRequested = false,
        autoStartedDone = false,
        requiredCloneWaitStartedAt = nil,
        acceptedRequestIds = {},
    }

    canHostStartMatch = function()
        if not isInDungeon() then return true end
        local record = cloneMemoryState.lowest()
        if record then
            local plr = getPlayerByNameInsensitive(record.name)
            if plr and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                return true
            end
            return false, "waiting for clone " .. record.name
        end
        local readyCount = getReadyDungeonCloneCounts()
        return readyCount > 0
    end

    local function accept(id, cloneName, gui)
        local reqId = tostring(id or "")
        if reqId == "" or hostState.acceptedRequestIds[reqId] then
            if gui then pcall(function() gui:Destroy() end) end
            return
        end
        if not fireRemote("respondJoinRequest", id, true) then return end
        if gui then pcall(function() gui:Destroy() end) end

        hostState.acceptedRequestIds[reqId] = true
        hostState.lastJoinActivityAt = os.clock()
        hostState.lobbyStartRequested = false
        hostState.autoStartedDone = false
        UI.addLog("Join request accepted: " .. tostring(cloneName or reqId), "success")
    end

    local showJoinRequest = getRemote("showJoinRequest", "RemoteEvent")
    if showJoinRequest then
        showJoinRequest.OnClientEvent:Connect(function(id, cloneName, act)
            if not runtimeActive() then return end
            if autoToggle and act ~= "close" and id then
                task.wait(0.1)
                if not runtimeActive() then return end
                local pg = localPlayer:FindFirstChild("PlayerGui")
                accept(id, cloneName, pg and pg:FindFirstChild("joinRequestConfirm_" .. id))
            end
        end)
    end

    task.spawn(function()
        while runtimeActive() and task.wait(0.1) do
            if not autoToggle then continue end
            local started = observeMatchState()
            if not started then
                hostState.requiredCloneWaitStartedAt = nil
                if hostState.lobbyCreatedSuccessfully then
                    local elapsed = os.clock() - hostState.lastJoinActivityAt
                    local remaining = math.max(0, CONFIG.START_DELAY - elapsed)
                    UI.setStatus("WAITING FOR CLONES", string.format("Lobby: %s (%s)", STAGES[currentStage].map, STAGES[currentStage].diff), "normal")
                    UI.setCountdown(string.format("Enter in %.1fs", remaining), true)
                    if elapsed >= CONFIG.START_DELAY and not hostState.lobbyStartRequested then
                        hostState.lobbyStartRequested = startMatch()
                    end
                else
                    UI.setStatus("LOBBY", "Preparing a new room", "normal")
                    UI.setCountdown("", false)
                end
            else
                if isDungeonBusy() then
                    hostState.autoStartedDone = true
                    UI.setCountdown("", false)
                    UI.setStatus("DUNGEON ACTIVE", string.format("%s (%s) | Target LV.%d", STAGES[currentStage].map, STAGES[currentStage].diff, STAGES[currentStage].targetLvl), "active")
                elseif not hostState.autoStartedDone then
                    local readyCount = getReadyDungeonCloneCounts()
                    local record = cloneMemoryState.lowest()

                    if record then
                        local plr = getPlayerByNameInsensitive(record.name)
                        local isReady = plr and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")

                        if isReady then
                            hostState.requiredCloneWaitStartedAt = nil
                            local elapsed = os.clock() - hostState.lastJoinActivityAt
                            local remaining = math.max(0, CONFIG.START_DELAY - elapsed)
                            UI.setStatus("CLONE READY", string.format("Synchronizing with %s", record.name), "normal")
                            UI.setCountdown(string.format("Start in %.1fs", remaining), true)
                            if elapsed >= CONFIG.START_DELAY then
                                hostState.autoStartedDone = startMatch()
                            end
                        else
                            hostState.requiredCloneWaitStartedAt = hostState.requiredCloneWaitStartedAt or os.clock()
                            local waitTime = os.clock() - hostState.requiredCloneWaitStartedAt
                            local remain = math.max(0, CONFIG.REQUIRED_CLONE_TIMEOUT - waitTime)
                            UI.setStatus("WAITING FOR CLONE", string.format("%s (LV.%s)", record.name, tostring(record.level)), "warning")
                            UI.setCountdown(string.format("Timeout %ds", math.floor(remain)), true)

                            if waitTime >= CONFIG.REQUIRED_CLONE_TIMEOUT then
                                UI.addLog(string.format("Clone %s timed out and was removed from memory", record.name), "warn")
                                cloneMemoryState.remembered[string.lower(record.name)] = nil
                                cloneMemoryState.save()
                                hostState.requiredCloneWaitStartedAt = nil
                            end
                        end
                    elseif readyCount > 0 then
                        local elapsed = os.clock() - hostState.lastJoinActivityAt
                        local remaining = math.max(0, CONFIG.START_DELAY - elapsed)
                        UI.setStatus("CLONES READY", string.format("%d clones in dungeon", readyCount), "normal")
                        UI.setCountdown(string.format("Start in %.1fs", remaining), true)
                        if elapsed >= CONFIG.START_DELAY then
                            hostState.autoStartedDone = startMatch()
                        end
                    end
                end
            end
        end
    end)

    task.spawn(function()
        local lastState = isMatchStarted()
        local activeStageIndex = currentStage
        local bossDetectedAt = nil

        while runtimeActive() and task.wait(1) do
            if not autoToggle then continue end
            local started = observeMatchState()

            if started and not lastState then
                hostState.isReplaying = false
                activeStageIndex = currentStage
                bossDetectedAt = nil
                UI.addLog(string.format("Dungeon started: %s (%s) | Target LV.%d",
                    STAGES[activeStageIndex].map, STAGES[activeStageIndex].diff, STAGES[activeStageIndex].targetLvl), "info")
            elseif not started and lastState then
                hostState.acceptedRequestIds = {}
                hostState.lobbyCreatedSuccessfully = false
                hostState.autoStartedDone = false
                bossDetectedAt = nil
                UI.addLog("Returned to lobby; preparing a new room", "info")
            end
            lastState = started

            if started then
                local busyNow = isDungeonBusy()
                local minCloneLvl = 999
                local cloneCount = 0

                for _, plr in ipairs(Players:GetPlayers()) do
                    if plr ~= localPlayer then
                        cloneCount = cloneCount + 1
                        local lvl = getPlayerLevel(plr)
                        if lvl > 0 and lvl < minCloneLvl then minCloneLvl = lvl end
                        local rName = (playerRealNames and playerRealNames[plr]) or tostring(plr.Name)
                        if lvl > 0 then
                            cloneMemoryState.remembered[string.lower(rName)] = { name = rName, level = lvl }
                        end
                    end
                end

                if busyNow then cloneMemoryState.save() end
                if cloneCount > 0 and minCloneLvl < 999 then
                    UI.setStage(activeStageIndex, minCloneLvl)
                end

                local activeStageData = STAGES[activeStageIndex] or STAGES[currentStage]
                local completionTarget = activeStageData.targetLvl
                if getDungeonProgress() == "bosskilled" then
                    bossDetectedAt = bossDetectedAt or os.clock()
                else
                    bossDetectedAt = nil
                end

                if isDungeonFinished() then
                    local levelPending = cloneCount > 0 and minCloneLvl == 999
                    if levelPending and os.clock() - (bossDetectedAt or os.clock()) < CONFIG.LEVEL_RESOLVE_GRACE then
                        matchCycle.finishConsumed = false
                        UI.setStatus("SYNCING CLONE LEVELS", "Waiting for replicated level data", "warning")
                    elseif cloneCount > 0 and minCloneLvl ~= 999 and minCloneLvl >= completionTarget then
                        UI.addLog(string.format("Boss defeated; clone level %d reached target %d", minCloneLvl, completionTarget), "success")
                        syncStageFromLowestCloneLevel(minCloneLvl)
                        cloneMemoryState.save()
                        leaveGame("Stage target reached")
                    elseif not hostState.isReplaying then
                        hostState.isReplaying = true
                        local reportedLevel = minCloneLvl == 999 and 0 or minCloneLvl
                        UI.addLog(string.format("Level below target (%d/%d); replaying", reportedLevel, completionTarget), "info")
                        local ok = fireRemote("replayDungeon", {{
                            dungeonProgress = "bossKilled", dungeonStarted = true, dungeonFinished = true,
                            hardcore = false, dungeonName = activeStageData.map, isHardcore = false, fightingBoss = true
                        }})
                        if not ok then
                            matchCycle.finishConsumed = false
                            UI.addLog("Replay request failed; retrying", "warn")
                        end
                        task.wait(CONFIG.REPLAY_COOLDOWN)
                        hostState.isReplaying = false
                    end
                end
            else
                if isInLobby() and not hostState.lobbyCreatedSuccessfully and os.clock() - hostState.lastLobbyCreateAt >= CONFIG.LOBBY_RETRY_DELAY then
                    local s = STAGES[currentStage]
                    hostState.lastLobbyCreateAt = os.clock()
                    local ok = createStageLobby(s.map, s.diff)
                    if ok then
                        hostState.lobbyCreatedSuccessfully = true
                        hostState.lobbyStartRequested = false
                        hostState.lastJoinActivityAt = os.clock()
                        UI.addLog("Lobby created: " .. s.map .. " (" .. s.diff .. ")", "success")
                    end
                end
            end
        end
    end)
end

function cloneMemoryState.runClone()
    local hostMissingSince = nil
    local lastJoinAttempt = -math.huge
    local isBuyingGamepass = false
    local isLeavingForGamepass = false
    local lastPriceFetch = -math.huge
    local lastPriceLevel = -1
    local passCache = {}
    local purchaseRetryAt = {}
    local purchaseFailures = {}

    local GAMEPASS_PRIORITY = {
        { id = "vip",               label = "VIP" },
        { id = "goldGamepass",      label = "X2 Gold" },
        { id = "extraItemGamepass", label = "+1 Item" },
        { id = "freeStatResets",    label = "Free Stat Resets" }
    }

    local function isPassOwned(id)
        local obj = localPlayer:FindFirstChild(id)
        if obj and obj:IsA("ValueBase") and obj.Value == true then
            return true
        end
        if passCache[id] and passCache[id].owned == true then
            return true
        end
        return false
    end

    local function updatePassPrices(force)
        local currentLevel = getLocalLevel()
        if currentLevel > 0 and currentLevel ~= lastPriceLevel then force = true end
        if not force and os.clock() - lastPriceFetch < 15 then return end
        lastPriceFetch = os.clock()
        if currentLevel > 0 then lastPriceLevel = currentLevel end

        for _, item in ipairs(GAMEPASS_PRIORITY) do
            if not runtimeActive() then return end
            local id = item.id
            if isPassOwned(id) then
                local knownCost = passCache[id] and passCache[id].cost or latestGamepassPrices[id] and latestGamepassPrices[id].cost
                passCache[id] = { cost = knownCost, owned = true, ready = true }
                latestGamepassPrices[id] = { cost = knownCost, owned = true, ready = true }
            elseif not passCache[id] or not passCache[id].cost or force then
                local ok, info = invokeRemote("getGoldGamepassPrice", id)
                if ok and type(info) == "table" then
                    passCache[id] = {
                        cost = tonumber(info.cost),
                        owned = (info.owned == true) or isPassOwned(id),
                        ready = info.ready ~= false,
                    }
                    latestGamepassPrices[id] = {
                        cost = passCache[id].cost,
                        owned = passCache[id].owned,
                        ready = passCache[id].ready,
                    }
                end
            end
        end
    end
    refreshGamepassPricesForHistory = function()
        updatePassPrices(true)
        return latestGamepassPrices
    end

    local function getNextPriorityTarget()
        for _, item in ipairs(GAMEPASS_PRIORITY) do
            if not isPassOwned(item.id) then
                local data = passCache[item.id]
                return item, data and data.ready and data.cost or nil
            end
        end
        return nil, nil
    end

    local goldPurchaseResult = getRemote("goldGamepassPurchaseResult", "RemoteEvent")
    if goldPurchaseResult then
        goldPurchaseResult.OnClientEvent:Connect(function(res)
            if not runtimeActive() then return end
            if type(res) == "table" and res.success and res.gamepassKey then
                if passCache[res.gamepassKey] then
                    passCache[res.gamepassKey].owned = true
                    latestGamepassPrices[res.gamepassKey] = {
                        cost = passCache[res.gamepassKey].cost or 0,
                        owned = true,
                        ready = true,
                    }
                end
                purchaseFailures[res.gamepassKey] = 0
                purchaseRetryAt[res.gamepassKey] = os.clock() + 5
            end
        end)
    end

    task.spawn(function()
        while runtimeActive() and task.wait(0.8) do
            if not autoToggle then continue end

            updatePassPrices(false)
            local currentGold = getLocalGold()
            local targetPass, cost = getNextPriorityTarget()

            for _, pItem in ipairs(GAMEPASS_PRIORITY) do
                local owned = isPassOwned(pItem.id)
                local isTarget = targetPass and (targetPass.id == pItem.id)
                local pCost = passCache[pItem.id] and passCache[pItem.id].cost or (isTarget and cost or 0)
                UI.updateGamepass(pItem.id, owned, currentGold, pCost, isTarget, isBuyingGamepass and isTarget)
            end

            local myLvl = getLocalLevel()
            if myLvl > 0 then
                UI.setStage(currentStage, myLvl)
            end

            if isInLobby() then
                isLeavingForGamepass = false

                if targetPass and cost and cost > 0 and currentGold >= cost and os.clock() >= (purchaseRetryAt[targetPass.id] or -math.huge) then
                    isBuyingGamepass = true
                    UI.setStatus("PURCHASING GAMEPASS", string.format("%s for %d Gold", targetPass.label, cost), "warning")
                    UI.addLog(string.format("Purchasing %s | Cost %d | Gold %d", targetPass.label, cost, currentGold), "info")

                    local bought = fireRemote("requestGoldGamepassPurchase", targetPass.id)
                    local confirmed = false
                    if bought then
                        local beforeGold = currentGold
                        local waitStart = os.clock()
                        repeat
                            task.wait(0.3)
                            currentGold = getLocalGold()
                        until currentGold < beforeGold or isPassOwned(targetPass.id) or (os.clock() - waitStart >= 6) or not isInLobby() or not runtimeActive()

                        confirmed = currentGold < beforeGold or isPassOwned(targetPass.id)
                        if confirmed then
                            if passCache[targetPass.id] then
                                passCache[targetPass.id].owned = true
                            end
                            UI.addLog(string.format("Purchased %s; %d Gold remaining", targetPass.label, currentGold), "success")
                        else
                            UI.addLog("Server did not confirm purchase: " .. targetPass.label, "warn")
                        end
                    end
                    if confirmed then
                        purchaseFailures[targetPass.id] = 0
                        purchaseRetryAt[targetPass.id] = os.clock() + 5
                    else
                        local failures = (purchaseFailures[targetPass.id] or 0) + 1
                        purchaseFailures[targetPass.id] = failures
                        purchaseRetryAt[targetPass.id] = os.clock() + math.min(60, 5 * (2 ^ (failures - 1)))
                    end
                    isBuyingGamepass = false
                    task.wait(CONFIG.GOLD_PURCHASE_DELAY or 1.5)
                else
                    isBuyingGamepass = false
                end
            else
                if targetPass and cost and currentGold >= cost then
                    if not isLeavingForGamepass then
                        isLeavingForGamepass = true
                        isBuyingGamepass = true
                        UI.setStatus("GAMEPASS AFFORDABLE", string.format("%d/%d Gold; returning for %s", currentGold, cost, targetPass.label), "warning")
                        UI.addLog(string.format("Returning to buy %s (%d/%d Gold)", targetPass.label, currentGold, cost), "info")
                        task.spawn(function()
                            leaveGame("Purchasing " .. targetPass.label)
                            task.wait(10)
                            isLeavingForGamepass = false
                        end)
                    end
                end
            end
        end
    end)

    local ls = localPlayer:FindFirstChild("leaderstats")
    local lvlObj = ls and ls:FindFirstChild("Level")
    if lvlObj then
        lvlObj:GetPropertyChangedSignal("Value"):Connect(function()
            if not runtimeActive() then return end
            updatePassPrices(true)
        end)
    end

    task.spawn(function()
        while runtimeActive() and task.wait(0.25) do
            if not autoToggle then continue end

            if isMatchStarted() then
                local hostPlr = getPlayerByNameInsensitive(HOST_NAME)
                if not hostPlr then
                    hostMissingSince = hostMissingSince or os.clock()
                    local missingFor = os.clock() - hostMissingSince
                    UI.setStatus("HOST CONNECTION LOST", string.format("Host missing for %.1f/%.1fs", missingFor, CONFIG.LOST_HOST_TIMEOUT), "danger")
                    UI.setCountdown(string.format("Exit in %.1fs", math.max(0, CONFIG.LOST_HOST_TIMEOUT - missingFor)), true)

                    if missingFor >= CONFIG.LOST_HOST_TIMEOUT then
                        UI.addLog("Host left; clone is returning", "warn")
                        leaveGame("Host left")
                        break
                    end
                else
                    hostMissingSince = nil
                    UI.setStatus("FOLLOWING HOST", string.format("Connected to %s", HOST_NAME), "active")
                    UI.setCountdown("", false)
                end
            else
                hostMissingSince = nil
                if isBuyingGamepass then
                    UI.setStatus("PURCHASING GAMEPASS", "Join requests paused", "warning")
                    UI.setCountdown("", false)
                else
                    local elapsed = os.clock() - lastJoinAttempt
                    local remaining = math.max(0, CONFIG.JOIN_REQUEST_INTERVAL - elapsed)
                    UI.setStatus("CONNECTING TO HOST", string.format("Target: %s", HOST_NAME), "normal")
                    UI.setCountdown(string.format("Request in %.1fs", remaining), true)

                    if elapsed >= CONFIG.JOIN_REQUEST_INTERVAL then
                        lastJoinAttempt = os.clock()
                        invokeRemote("sendJoinRequest", HOST_NAME)
                    end
                end
            end
        end
    end)

    Players.PlayerRemoving:Connect(function(plr)
        if not runtimeActive() then return end
        local rName = (playerRealNames and playerRealNames[plr]) or tostring(plr.Name)
        if autoToggle and isMatchStarted() and string.lower(rName) == string.lower(HOST_NAME) then
            UI.addLog("Host disconnect detected", "warn")
            leaveGame("Host disconnected")
        end
    end)
end

task.spawn(function()
    local ok, err = pcall(function()
        if isHost then
            cloneMemoryState.runHost()
        else
            cloneMemoryState.runClone()
        end
    end)
    if not ok then
        reportRemoteProblem("runtime", tostring(err))
        if UI then UI.addLog("Runtime initialization failed: " .. tostring(err), "danger") end
    end
end)

pcall(function()
    UI.addLog("Auto Carry Pro initialized", "success")
    if not isHost then
        UI.addLog("Clone automation ready", "info")
    end
end)
print("[AutoCarry Pro] Initialized")
